# D4 실전 해설

## P01. 터널 탈출

현재 터널이 해당 방향으로 열려 있어야 하고, 다음 터널도 반대 방향으로 열려 있어야 이동할 수 있다. 시간 제한이 있으므로 BFS에서 현재 시간이 L보다 작을 때만 다음 칸을 넣는다.

```python
from collections import deque

dr = [-1, 1, 0, 0]
dc = [0, 0, -1, 1]
opposite = [1, 0, 3, 2]
connect = {
    0: [],
    1: [0, 1, 2, 3],
    2: [0, 1],
    3: [2, 3],
    4: [0, 3],
    5: [1, 3],
    6: [1, 2],
    7: [0, 2],
}

T = int(input())
for tc in range(1, T + 1):
    N, M, R, C, L = map(int, input().split())
    board = [list(map(int, input().split())) for _ in range(N)]
    visited = [[0] * M for _ in range(N)]
    q = deque([(R, C)])
    visited[R][C] = 1

    while q:
        r, c = q.popleft()
        if visited[r][c] == L:
            continue
        pipe = board[r][c]
        for d in connect[pipe]:
            nr = r + dr[d]
            nc = c + dc[d]
            if 0 <= nr < N and 0 <= nc < M and visited[nr][nc] == 0:
                next_pipe = board[nr][nc]
                if opposite[d] in connect[next_pipe]:
                    visited[nr][nc] = visited[r][c] + 1
                    q.append((nr, nc))

    ans = sum(1 for r in range(N) for c in range(M) if visited[r][c])
    print(f"#{tc} {ans}")
```

## P02. 등산로 조성

모든 최고점에서 DFS를 시작한다. 다음 칸이 더 낮으면 그냥 이동한다. 더 낮지 않더라도 아직 깎기 기회를 쓰지 않았고, K만큼 깎으면 현재보다 낮아질 수 있다면 그 칸을 `현재 높이 - 1`로 임시 변경하고 이동한다. DFS가 끝나면 반드시 원래 높이로 복구한다.

```python
dr = [-1, 1, 0, 0]
dc = [0, 0, -1, 1]

T = int(input())
for tc in range(1, T + 1):
    N, K = map(int, input().split())
    board = [list(map(int, input().split())) for _ in range(N)]
    max_height = max(max(row) for row in board)
    visited = [[False] * N for _ in range(N)]
    ans = [0]

    def dfs(r, c, length, used_cut):
        ans[0] = max(ans[0], length)

        for d in range(4):
            nr = r + dr[d]
            nc = c + dc[d]
            if not (0 <= nr < N and 0 <= nc < N):
                continue
            if visited[nr][nc]:
                continue

            if board[nr][nc] < board[r][c]:
                visited[nr][nc] = True
                dfs(nr, nc, length + 1, used_cut)
                visited[nr][nc] = False
            elif not used_cut and board[nr][nc] - K < board[r][c]:
                original = board[nr][nc]
                board[nr][nc] = board[r][c] - 1
                visited[nr][nc] = True
                dfs(nr, nc, length + 1, True)
                visited[nr][nc] = False
                board[nr][nc] = original

    for r in range(N):
        for c in range(N):
            if board[r][c] == max_height:
                visited[r][c] = True
                dfs(r, c, 1, False)
                visited[r][c] = False

    print(f"#{tc} {ans[0]}")
```

## P03. 프로세서 연결

각 내부 프로세서마다 네 방향 연결을 시도한다. 전선을 놓을 수 있으면 표시하고 재귀로 들어간 뒤 되돌린다. 연결하지 않는 선택도 필요하다. 남은 프로세서를 모두 연결해도 현재 최고 연결 수를 넘지 못하면 가지치기한다.

```python
dr = [-1, 1, 0, 0]
dc = [0, 0, -1, 1]

def is_edge(r, c, N):
    return r == 0 or r == N - 1 or c == 0 or c == N - 1

T = int(input())
for tc in range(1, T + 1):
    N = int(input())
    board = [list(map(int, input().split())) for _ in range(N)]
    cores = []
    for r in range(N):
        for c in range(N):
            if board[r][c] == 1 and not is_edge(r, c, N):
                cores.append((r, c))

    best_connected = [0]
    best_wire = [10**9]

    def path_cells(r, c, d):
        cells = []
        nr = r + dr[d]
        nc = c + dc[d]
        while 0 <= nr < N and 0 <= nc < N:
            if board[nr][nc] != 0:
                return None
            cells.append((nr, nc))
            nr += dr[d]
            nc += dc[d]
        return cells

    def dfs(idx, connected, wire):
        if connected + (len(cores) - idx) < best_connected[0]:
            return
        if idx == len(cores):
            if connected > best_connected[0]:
                best_connected[0] = connected
                best_wire[0] = wire
            elif connected == best_connected[0]:
                best_wire[0] = min(best_wire[0], wire)
            return

        r, c = cores[idx]
        for d in range(4):
            cells = path_cells(r, c, d)
            if cells is None:
                continue
            for wr, wc in cells:
                board[wr][wc] = 2
            dfs(idx + 1, connected + 1, wire + len(cells))
            for wr, wc in cells:
                board[wr][wc] = 0

        dfs(idx + 1, connected, wire)

    dfs(0, 0, 0)
    print(f"#{tc} {best_connected[0]} {best_wire[0]}")
```

위 문제는 구현 실수 비율이 높다. 실제 풀이에서는 `path_cells`, 전선 표시, 전선 복구를 분리하는 것이 핵심이다.

## P04. 보급로 최소비용

칸마다 이동 비용이 다르므로 일반 BFS가 아니라 다익스트라를 사용한다. 우선순위 큐에서 현재까지 비용이 가장 작은 칸을 먼저 꺼낸다.

```python
import heapq

dr = [-1, 1, 0, 0]
dc = [0, 0, -1, 1]
INF = 10**9

T = int(input())
for tc in range(1, T + 1):
    N = int(input())
    board = [list(map(int, input().strip())) for _ in range(N)]
    dist = [[INF] * N for _ in range(N)]
    dist[0][0] = 0
    heap = [(0, 0, 0)]

    while heap:
        cost, r, c = heapq.heappop(heap)
        if cost != dist[r][c]:
            continue
        for d in range(4):
            nr = r + dr[d]
            nc = c + dc[d]
            if 0 <= nr < N and 0 <= nc < N:
                next_cost = cost + board[nr][nc]
                if next_cost < dist[nr][nc]:
                    dist[nr][nc] = next_cost
                    heapq.heappush(heap, (next_cost, nr, nc))

    print(f"#{tc} {dist[N - 1][N - 1]}")
```
